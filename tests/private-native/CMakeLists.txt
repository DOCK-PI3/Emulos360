cmake_minimum_required(VERSION 3.24)
project(EmulosPrivateTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Qml Network Test)
qt_standard_project_setup()
qt_add_executable(private_native_tests private_native_tests.cpp ../../app/private_netplay.cpp ../../app/private_netplay.h)
target_link_libraries(private_native_tests PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Network Qt6::Test)
target_include_directories(private_native_tests PRIVATE ../../app)
if(NOT MSVC)
  target_compile_options(private_native_tests PRIVATE -Wall -Wextra -Wpedantic -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(private_native_tests PRIVATE -fsanitize=address,undefined)
endif()
