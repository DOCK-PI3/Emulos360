cmake_minimum_required(VERSION 3.24)
project(EmulosCoreMaintenance LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(PROJECT_ROOT "../.." ABSOLUTE)
set(CORE_ROOT "${PROJECT_ROOT}/engine/xenia" CACHE PATH "Actual core under test")
option(MAINTENANCE_SANITIZERS "Enable ASan and UBSan" OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_fragments.py"
    "${CORE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}" COMMAND_ERROR_IS_FATAL ANY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CORE_ROOT}/src/xenia/kernel/xam/xam_net.cc")
add_executable(core_maintenance_tests netplay_contract_tests.cpp)
target_include_directories(core_maintenance_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_include_directories(core_maintenance_tests SYSTEM PRIVATE "${CORE_ROOT}/src" "${CORE_ROOT}/third_party")
if(MSVC)
  target_compile_options(core_maintenance_tests PRIVATE /W4 /WX /utf-8 /arch:AVX)
else()
  target_compile_options(core_maintenance_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -mavx)
  if(MAINTENANCE_SANITIZERS)
    target_compile_options(core_maintenance_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(core_maintenance_tests PRIVATE -fsanitize=address,undefined)
  endif()
endif()
enable_testing()
add_test(NAME netplay_actual_fragments COMMAND core_maintenance_tests)
