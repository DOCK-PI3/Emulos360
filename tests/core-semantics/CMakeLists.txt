cmake_minimum_required(VERSION 3.24)
project(EmulosCoreSemantics LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(PROJECT_ROOT "../.." ABSOLUTE)
set(CORE_ROOT "${PROJECT_ROOT}/engine/xenia" CACHE PATH "Core headers and dependencies")
set(CORE_VALUE_SOURCE "${CORE_ROOT}/src/xenia/cpu/hir/value.cc" CACHE FILEPATH "Actual Value implementation under test")
option(CORE_SEMANTICS_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
add_library(core_value_under_test STATIC "${CORE_VALUE_SOURCE}")
target_include_directories(core_value_under_test SYSTEM PRIVATE "${CORE_ROOT}" "${CORE_ROOT}/src" "${CORE_ROOT}/third_party")
add_executable(core_semantics_tests core_semantics_tests.cpp)
target_include_directories(core_semantics_tests SYSTEM PRIVATE "${CORE_ROOT}" "${CORE_ROOT}/src" "${CORE_ROOT}/third_party")
if(MSVC)
  # MSVC requires unresolved references from unused members of an extracted
  # library object to be resolved before /OPT:REF. Compile the entire real
  # translation unit, but link a byte-for-byte slice of the tested member.
  # This supplies no fake definitions for unrelated core subsystems.
  file(READ "${CORE_VALUE_SOURCE}" value_source)
  string(FIND "${value_source}" "void Value::Extract(" extract_start)
  string(FIND "${value_source}" "void Value::Permute(" extract_end)
  if(extract_start LESS 0 OR extract_end LESS extract_start)
    message(FATAL_ERROR "Unable to locate the original Extract function")
  endif()
  math(EXPR extract_size "${extract_end} - ${extract_start}")
  string(SUBSTRING "${value_source}" ${extract_start} ${extract_size} extract_body)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/extract_under_test.cc"
      "#include \"xenia/cpu/hir/value.h\"\nnamespace xe::cpu::hir {\n${extract_body}\n}\n")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CORE_VALUE_SOURCE}")
  add_library(core_extract_under_test STATIC "${CMAKE_CURRENT_BINARY_DIR}/extract_under_test.cc")
  target_include_directories(core_extract_under_test SYSTEM PRIVATE "${CORE_ROOT}" "${CORE_ROOT}/src" "${CORE_ROOT}/third_party")
  target_compile_options(core_extract_under_test PRIVATE /W4 /WX /utf-8 /arch:AVX /wd4201 /wd4324 /wd4100)
  target_link_libraries(core_semantics_tests PRIVATE core_extract_under_test)
  add_dependencies(core_semantics_tests core_value_under_test)
  target_compile_options(core_value_under_test PRIVATE /W4 /Gy /Gw /utf-8 /arch:AVX /wd4201 /wd4324 /wd4100 /wd4244 /wd4267 /wd4458)
  target_compile_options(core_semantics_tests PRIVATE /W4 /WX /utf-8 /arch:AVX /wd4201 /wd4324 /wd4100)
  target_compile_definitions(core_value_under_test PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_link_options(core_semantics_tests PRIVATE /OPT:REF)
else()
  target_link_libraries(core_semantics_tests PRIVATE core_value_under_test)
  foreach(target core_value_under_test core_semantics_tests)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -mavx -ffunction-sections -fdata-sections)
  endforeach()
  target_compile_options(core_semantics_tests PRIVATE -Werror)
  # The entire real Value translation unit is compiled. The linker removes
  # unrelated HIR operations; no fake HIRBuilder, VFS, kernel or JIT is supplied.
  target_link_options(core_semantics_tests PRIVATE -Wl,--gc-sections)
  if(CORE_SEMANTICS_SANITIZERS)
    foreach(target core_value_under_test core_semantics_tests)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    endforeach()
    target_link_options(core_semantics_tests PRIVATE -fsanitize=address,undefined)
  endif()
endif()
enable_testing()
add_test(NAME extract_valid_lanes COMMAND core_semantics_tests --lanes)
add_test(NAME extract_all_indices COMMAND core_semantics_tests --all-indices)
