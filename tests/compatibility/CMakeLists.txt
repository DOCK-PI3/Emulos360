cmake_minimum_required(VERSION 3.24)
project(EmulosCompatibilityTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Qml Network Test)
get_filename_component(ROOT "../.." ABSOLUTE)
qt_add_executable(compatibility_tests compatibility_tests.cpp
    ${ROOT}/app/game_compatibility.cpp ${ROOT}/app/game_compatibility.h)
target_include_directories(compatibility_tests PRIVATE ${ROOT}/app)
target_link_libraries(compatibility_tests PRIVATE Qt6::Qml Qt6::Network Qt6::Test)
qt_add_resources(compatibility_tests catalog BASE ${ROOT} PREFIX "/" FILES ${ROOT}/app/compatibility-canary.json)
if(NOT MSVC)
    target_compile_options(compatibility_tests PRIVATE -Wall -Wextra -Wpedantic -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(compatibility_tests PRIVATE -fsanitize=address,undefined)
endif()
